45async def breaker_call_async(breaker: pybreaker.CircuitBreaker, coro_func, *args, **kwargs): 

46 """asyncio-native circuit breaker call. 

47 

48 pybreaker's built-in call_async uses Tornado's gen.coroutine which is 

49 unavailable in asyncio-only environments. This wrapper replicates the 

50 same state machine but awaits the coroutine directly. 

51 """ 

52 with breaker._lock: 

53 state = breaker.state 

54 # before_call raises CircuitBreakerError if the circuit is open 

55 state.before_call(coro_func, *args, **kwargs) 

56 for listener in breaker.listeners: 

57 listener.before_call(breaker, coro_func, *args, **kwargs) 

58 

59 try: 

60 result = await coro_func(*args, **kwargs) 

61 except BaseException as exc: 

62 with breaker._lock: 

63 state._handle_error(exc) 

64 raise 

65 else: 

66 with breaker._lock: 

67 state._handle_success() 

68 return result
