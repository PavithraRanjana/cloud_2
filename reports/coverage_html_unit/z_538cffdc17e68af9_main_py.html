69def resolve_seat(seat_preference: str | None) -> tuple[str, str]: 

70 seat = (seat_preference or "").strip().split(",")[0].strip() 

71 if not seat: 

72 # Safety fallback — should not occur; all bookings now require a seat 

73 row = random.randint(1, 35) 

74 col = random.choice(["A", "B", "C", "D", "E", "F"]) 

75 seat = f"{row}{col}" 

76 row_str = "".join(c for c in seat if c.isdigit()) 

77 row = int(row_str) if row_str else 15 

78 group = "A" if row <= 10 else "B" if row <= 20 else "C" 

79 return seat, group
